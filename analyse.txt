Échec de l’actualisation Jikan : HTTP 504 en clickant desus

Oui, j'ai identifié un problème dans ma V1 : l'endpoint Jikan que j'avais utilisé était incorrect.

J'avais mis :

/v4/anime/season/2026/summer

alors que l'endpoint saisonnier Jikan v4 est :

/v4/seasons/2026/summer

La documentation/les clients Jikan utilisent bien cette forme /seasons/{year}/{season}.

Il y a également un deuxième point : Jikan connaît actuellement des problèmes de 504 côté serveur, avec des signalements récents de 504 sur des requêtes qui doivent récupérer les données auprès de MAL.

J'ai donc corrigé la V1 avec :

✅ endpoint correct /seasons/...
✅ retry automatique en cas de 429
✅ retry automatique en cas de 500 / 502 / 503 / 504
✅ backoff progressif entre les tentatives
✅ pagination basée sur has_next_page
✅ message d'erreur plus explicite
✅ aucun impact sur ta BDD locale

function openDB(){
  return new Promise((resolve,reject)=>{
    const r=indexedDB.open(DB_NAME,DB_VERSION);
    r.onupgradeneeded=()=>{

      if(!d.objectStoreNames.contains("anime"))d.createObjectStore("anime",{keyPath:"mal_id"});

function store(name,mode="readonly"){return db.transaction(name,mode).objectStore(name)}
function put(name,v){return new Promise((res,rej)=>{const r=store(name,"readwrite").put(v);r.onsuccess=()=>res();r.onerror=()=>rej(r.error)})}
function get(name,k){return new Promise((res,rej)=>{const r=store(name).get(k);r.onsuccess=()=>res(r.result);r.onerror=()=>rej(r.error)})}
function all(name){return new Promise((res,rej)=>{const r=store(name).getAll();r.onsuccess=()=>res(r.result);r.onerror=()=>rej(r.error)})}

async function fetchJSON(url, attempts=4){
  let lastError;
  for(let attempt=1;attempt<=attempts;attempt++){
    try{
      const response=await fetch(url,{cache:"no-store"});
      const text=await response.text();
      let body=null;
      try{body=text?JSON.parse(text):null}catch{}
      if(response.ok)return body;
      const serverMessage=body?.message?` — ${body.message}`:"";
      const error=new Error(`HTTP ${response.status}${serverMessage}`);
      error.status=response.status;
      lastError=error;
      if(![408,429,500,502,503,504].includes(response.status)||attempt===attempts)throw error;
    }catch(error){
      lastError=error;
      if(attempt===attempts)throw error;
    }
    await sleep(1200*Math.pow(2,attempt-1)+Math.floor(Math.random()*500));
  }
  throw lastError||new Error("Erreur réseau");
}

async function refreshSeason(year,season){
  if($("#refreshBtn").disabled)return;
  setStatus(`Test/synchronisation Jikan : ${year} ${seasonNames[season]}…`,"");
  $("#refreshBtn").disabled=true; $("#testBtn").disabled=true;

  const seasonId=seasonKey(year,season);
  const previousLinks=(await all("seasons")).filter(x=>x.seasonKey===seasonId);
  const newIds=[];

  try{
    // Important: use the canonical v4 seasonal endpoint without sfw=false.
    // The public Jikan instance has had intermittent 504s when its MAL backend fails.
    let page=1, hasNext=true, count=0;
    while(hasNext){
      setStatus(`Synchronisation Jikan : page ${page}… (${count} anime reçus)`);
      const data=await fetchJSON(`${API_SEASON}/${year}/${season}?page=${page}&limit=25`);
      const items=data?.data||[];
      if(!items.length && page===1)throw new Error("Jikan a renvoyé une saison vide.");

      for(const raw of items){
        const anime=normalize(raw);
        await put("anime",anime);
        newIds.push(anime.mal_id);
        count++;
      }
      hasNext=Boolean(data?.pagination?.has_next_page);
      page++;
      if(hasNext)await sleep(900);
    }

    // Replace the local membership of this season only after the complete
    // download succeeded. A failed refresh therefore cannot corrupt the cache.
    const keep=new Set(newIds);
    for(const old of previousLinks){
      if(!keep.has(old.mal_id))await remove("seasons",old.key);
    }
    for(const id of newIds){
      await put("seasons",{key:`${seasonId}:${id}`,seasonKey:seasonId,year,season,mal_id:id});
    }
    await put("meta",{key:seasonId,lastRefresh:new Date().toISOString(),count});
    setStatus(`✓ Synchronisation terminée : ${count} anime enregistrés localement.`,"ok");
    await loadSeason();
  }catch(e){
    const hint=e.status===504
      ?"Jikan renvoie actuellement un 504 : le serveur public n'arrive pas à joindre MyAnimeList. La BDD locale n'a pas été modifiée."
      :"La synchronisation a échoué. La BDD locale précédente est conservée.";
    setStatus(`Échec : ${e.message}. ${hint}`,"error");
  }finally{
    $("#refreshBtn").disabled=false; $("#testBtn").disabled=false;
  }
}

async function importData(file){

  if(p?.format!=="anime-history"||!Array.isArray(p.anime)||!Array.isArray(p.seasons)||!Array.isArray(p.user))throw Error("Fichier Anime History invalide.");

  setStatus(`✓ Import terminé : ${p.anime.length} anime, ${p.seasons.length} associations de saisons, ${p.user.length} données utilisateur.`,"ok");
  await loadSeason();
}

readme = r"""# Anime History — V1.2

## 1. Objectif du projet

Anime History est une petite application personnelle, locale et orientée « historique anime ».

L'idée est de conserver une copie locale des informations publiques de MyAnimeList organisées par :

- année ;
- saison : Winter / Spring / Summer / Fall ;
- anime appartenant à chaque saison.

La source distante n'est PAS utilisée pour le fonctionnement quotidien du site.

Le principe est :

    MyAnimeList
         ↑
      Jikan
         ↑
    synchronisation
         │
         ▼
    IndexedDB locale
         │
         ├── données anime
         ├── appartenance aux saisons
         └── métadonnées de synchronisation

    Données personnelles
         │
         ├── vu / non vu
         ├── favori / non favori
         ├── classement personnel (prévu)
         └── notes / tags personnels (prévu)

Le projet ne cherche donc pas à remplacer MyAnimeList en ligne.
Il sert plutôt de « mémoire personnelle » permettant de revoir l'histoire de ses anime saison par saison.

---

## 2. Pourquoi Jikan ?

Jikan est une API REST open-source non officielle qui expose des données publiques de MyAnimeList.

Le projet utilise Jikan uniquement comme couche de synchronisation.

Jikan n'est jamais considéré comme la base de données de l'application.

C'est important pour la stabilité :

- si Jikan est indisponible, le site continue à fonctionner avec les données déjà synchronisées ;
- les clics « Vu » et « Favori » ne dépendent pas du réseau ;
- les recherches ne dépendent pas du réseau ;
- les vues par année ne dépendent pas du réseau ;
- les données utilisateur ne sont jamais envoyées à Jikan.

Documentation Jikan :
https://docs.jikan.moe/

Projet Jikan REST :
https://github.com/jikan-me/jikan-rest

---

## 3. Pourquoi IndexedDB ?

Le navigateur dispose de plusieurs mécanismes de stockage.

localStorage est pratique mais limité et peu adapté à une véritable base de données.

IndexedDB est donc utilisé comme base locale.

Stores utilisés :

## 8. Important : disponibilité de Jikan

Jikan est un service tiers et n'est pas contrôlé par cette application.

Jikan fonctionne en récupérant/parsing des données publiques de MyAnimeList.

Il peut donc arriver que :

    Jikan -> MyAnimeList

échoue alors que :

    Chrome -> MyAnimeList

fonctionne parfaitement.

Dans ce cas, l'application locale n'est pas en panne.

Le bouton `Tester Jikan` permet de distinguer les deux situations.

L'application est volontairement conçue pour que l'indisponibilité de Jikan ne détruise jamais la base locale.

---

## 9. Pourquoi ne pas récupérer directement MyAnimeList depuis le HTML ?

Le projet évite volontairement de faire du scraping direct de MAL depuis le navigateur.

Cela poserait notamment :

- problèmes CORS ;
- dépendance au HTML de MAL ;
- changements de structure du site ;
- gestion plus compliquée des erreurs ;
- risque de casser l'application lorsque MAL modifie ses pages.

Jikan fournit précisément une couche REST pour ce type d'accès.
